Tutorial: Membuat Galeri dengan Bootstrap Grid

galeri.html 
<!-- Grid Container -->
<div class="row g-3">

  <!-- Kolom biasa (1/3 lebar di lg) -->
  <div class="col-md-6 col-lg-4">
    <div class="gallery-item position-relative">
      <img src="foto.jpg" class="img-fluid rounded">
      <!-- Overlay hover -->
      <div class="gallery-overlay">...</div>
    </div>
  </div>

  <!-- Kolom lebar (2/3 di lg, featured) -->
  <div class="col-md-6 col-lg-8">
    ...
  </div>

</div>
